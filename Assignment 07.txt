public class BrowserHistory{
  public int count=0;
  public String [] arr ;
  public  BrowserHistory(int n){
    arr = new String [n];
  }
  public void showHistory(){
    if(count>0){
      System.out.println("Browser History:");
      for(int i =0;i<count;i++){
        System.out.println("Visited: "+arr[i]);
      }
    }
    else{
      System.out.println("No web pages visited yet.");
    }
  }
  public void currentWeb(){
    if(count>0){
      System.out.println("Current web page: "+arr[count-1]);
    }
    else{
      System.out.println("You have not visited any website yet!");
    }
  }
  public void visitWeb(String s){
    if(count<arr.length){
    arr[count++]=s;
    System.out.println("Visited: "+s);
    }
    else{
      System.out.println("History is full. Cannot visit "+s);
    }
  }
  public void back(){
    arr[--count] =null;
  }
}

public class CustomerService{
    public int count =0;
    public int count1=0;
    public int next_empty;
    public int next_serve;
    public String arr[];
    public int len;
    public CustomerService(int n){
      len =n;
      arr=new String [len];
    }
    public void addCustomer(String s){
      if(count<len){
      next_empty =count1%len;
      arr[next_empty]=s;
      count1++;
      count++;
      System.out.println("Added: "+s);
      }
      else{
         System.out.println("Queue is full. Cannot add "+s);
      }
    }
    public void serveCustomer(){
      if(count==0){
        System.out.println("No customers to serve");
      }
      else{
      next_serve =next_serve%len;
        System.out.println("Serving "+arr[next_serve]);
        arr[next_serve]=null;
        next_serve++;
        count--;
      }
    }
  }

public class Player extends SportsPerson{
  public int goals;
  public int matches;
  public double ratio;
  public Player(String team, String name,String type,int goals,int matches){
    super(team,name,type);
    this.goals=goals;
    this.matches=matches;
    super.earningPerMatch = (goals * 1000) + (matches * 10);
  }
  public void calculateRatio(){
    double n =goals;
    ratio =n/matches;
  }
  public void printDetails(){
    System.out.println(getNameTeam()+"\nTeam Role: "+super.role+"\nTotal Goal: "+goals+", Total Played: "+matches+"\nGoal Ratio: "+String.format("%.2f",ratio)+"\nMatch Earning: "+super.earningPerMatch+"k");
  }
}
