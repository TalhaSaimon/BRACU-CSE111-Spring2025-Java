

public class Person{
  public String name;
  public int age;
  public String type;
  public boolean dose1=false;
  public boolean dose2=false;
  public Vaccine  c;
  public Person(String s1,int n1,String s2){
    name=s1;
    age=n1;
    type=s2;
  }
   public Person(String s1,int n1){
    name=s1;
    age=n1;
  }
  public void pushVaccine(Vaccine v){
    if(dose1==false){
      if(type.equals("Student")){
      dose1 =true;
      System.out.println("1st dose done for "+name);
      c=v;
      }
      else if(age>=25){
        dose1 =true;
        System.out.println("1st dose done for "+name);
        c=v;
      }
      else{
        System.out.println("Sorry "+name+", Minimum age for taking vaccines is 25 years now.");
      }
    }
  }
  public void pushVaccine(Vaccine v,String s){
    if(s.equals("1st Dose") && dose1==false){
      pushVaccine(v);
    }
    else if(dose1){
      System.out.println("Sorry "+name+", invalid dose request.");
    }
    else if(dose2){
      System.out.println("Sorry "+name+", you already received both doses.");
    }
    else{
      if(c.getName().equals(v.getName())){
        dose2 =true;
        System.out.println("2nd dose done for "+name);
      }
      else{
        System.out.println("Sorry "+name+", you can’t take 2 different vaccines");
      }
      
    }
  }
  public void showDetail(){
    String s1="";
    String s2="";
    if(dose1){
      s1="Given";
      if(dose2){
      s2="Given";
      }
      else{
        s2 ="Please come after "+c.getDays()+" days";
      }
    }
    else{
      s1 ="Please come after "+c.getDays()+" days";
    }
    System.out.println("Name: "+name+" Age: "+age+" Type: "+type+"\nVaccine name: "+c.getName()+"\n1st dose: "+s1+"\n2nd dose: "+s2);
  }
}

public class NikeBD{

  public int soldLocal=0;
  public NikeBD(String name){
    this.name=name;
    count++;
  }
  public static void status(){
    System.out.print("Nike Bangladesh Status:\nBranches Opened: "+count+"\nCurrently Stocked: ");
    for(int i=0;i<3;i++){
      if(i!=2){
        System.out.print(arr1[i]+": "+arr2[i]+", ");
      }
      else{
         System.out.println(arr1[i]+": "+arr2[i]);
      }
    }
    System.out.println("Sold: "+sold);
  }
  public void restockProducts(String s,int n){
    for(int  i=0;i<3;i++){
      if(s.equals(arr1[i])){
        qty[i]+=n;
        arr2[i]+=n;
        break;
      }
    }
  }
  public void restockProducts(String [] products,int[] qty){
     for(int  i=0;i<3;i++){
    restockProducts(products[i],qty[i]);
     }
  }
  public void details(){
    System.out.print("Nike "+name+" outlet:\nProducts Currently Stocked: ");
    for(int i=0;i<3;i++){
      if(i!=2){
        System.out.print(products[i]+": "+qty[i]+", ");
      }
      else{
         System.out.println(products[i]+": "+qty[i]);
      }
    }
    System.out.println("Sold: "+soldLocal);
  }
  public void productSold(String s,int n){
    for(int i=0;i<3;i++){
      if(s.equals(products[i])){
        soldLocal+=n;
        qty[i] -= n;
        arr2[i] -= n;
        sold +=n;
        break;
      }
    }
  }
  public void productSold(String s1,int n1,String s2,int n2){
    productSold( s1, n1);
    productSold(s2, n2);
  }
}
