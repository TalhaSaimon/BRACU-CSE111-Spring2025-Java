    name =s;
    price =discountCalculate(n);
    total_revenue +=price;
    Book.total_books_sold++;
  }
  public double discountCalculate(int n){
    double dis_price= 150.0*(100-n)/100;
    return dis_price;
  }
  public void bookDetails(){
    System.out.println("Title: "+name+"\nPrice after Discount: "+price+" TK");
  }
}

public class Borrower{
  public static int book_count[] = {3, 3, 3};
  public static String book_name[] = {"Pather Panchali", "Durgesh Nandini", "Anandmath"};
  public String name;
  public String []borrow_history=new String [9];
  public int idx=0;
  
  
  
  public Borrower(String s){
    name =s;
  }
  public static void bookStatus(){
    System.out.println("Available Books: ");
    for(int i=0;i<3;i++){
      System.out.println(Borrower.book_name[i]+": "+Borrower.book_count[i]);
    }
  }
  public void borrowBook(String s){
    for(int i=0;i<3;i++){
      if(s.equals(Borrower.book_name[i])){
        if(Borrower.book_count[i]>0){
          Borrower.book_count[i] -=1;
          borrow_history[idx++]= Borrower.book_name[i];
        }
        else{
          System.out.println("This book is not available.");
        }
         break;
      }
     
    }
  }
  public void borrowerDetails(){
    System.out.println("Name: "+name+"\nBooks Borrowed:");
    for(int i=0;i<idx;i++){
       System.out.println(borrow_history[i]);
    }
  }
  public static int remainingBooks(String s){
    for(int i=0;i<3;i++){
      if(s.equals(Borrower.book_name[i])){
        return Borrower.book_count[i];
      }
    }
    return 0;
  }
}

public class Cargo{
  private static double capacity =10.0;
  public static int count=0;
  public int id;
  public String content;
  public double weight;
  public boolean loaded =false;
  public Cargo(String s, double n){
    content =s;
    weight =n;
    id =++count;
  }
  public void details(){
    System.out.println("Cargo ID: "+id+", Contents: "+content+", Weight: "+weight+", Loaded: "+loaded);
  }
  public void load(){
    if(capacity>=weight){
      capacity-=weight;
      loaded =true;
      System.out.println("Cargo "+id+" loaded for transport.");
    }
    else{
      System.out.println("Cannot load cargo, exceeds weight capacity.");
    }
  }
  public void unload(){
      capacity+=weight;
      loaded =false;
      System.out.println("Cargo "+id+" unloaded.");
  }
  public static String capacity(){
    return  String.format("%.1f",capacity);
  }
}
