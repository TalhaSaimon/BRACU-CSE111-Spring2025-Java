public class Manager extends Employee{
  public double bonus ;
  public double finalSalary ;
  public int increment;
  public Manager(String name, double n1, int n2,int bonus){
    super(name,n1,n2);
    this.bonus =bonus;
  }
  public void displayInfo(){
    super.displayInfo();
    System.out.println("Bonus: "+bonus+"%\nFinal Salary: $"+finalSalary);
  }
  public void calculateSalary(){
    double n =getBaseSalary();
    finalSalary =n+((n*bonus)/100);
  }
  public void requestIncrement(int n){
    int h =getHoursWorked();
    if(h<80){
      System.out.println("Increment denied.");
    }
    else{
      if(h<100){
        increment =n/2;
      }
      else{
        increment =n;
      }
      setBaseSalary(getBaseSalary()+increment);
      System.out.println("$"+increment+" Increment approved.");
    }
  }
}

    super(movie,date,showtime,0.0);
    this.genre=genre;
    seat =seatTypes[0];
    CinemexTicket.count++;
    ticketId =movie+"-"+seat.charAt(0)+"-"+CinemexTicket.count;
  }
  public CinemexTicket(String movie,String showtime,String genre,String date,String seat){
    super(movie,date,showtime,0.0);
    this.genre=genre;
    this.seat =seat;
    
    CinemexTicket.count++;
    ticketId =movie+"-"+seat.charAt(0)+"-"+CinemexTicket.count;
  }
  

  public void calculateTicketPrice(){
    String[] arr = showtime.split(":");
    double n1 =Integer.parseInt(arr[0]);
    double n2 =Integer.parseInt(arr[1]);
    double n =n1+n2/60;
    for(int i=0;i<seatTypes.length;i++){
      if(seat.equals(seatTypes[i])){
        setPrice(seatPrices[i]);
      }
    }
    if(18.00<=n && 23.00>=n){
      double price = (getPrice() * nightShowCharge / 100.0)+ getPrice() ;
      setPrice(price);
    }
    
    
    System.out.println("Ticket price is calculated successfully.");
  }
  public String confirmPayment(){
    if(flag==true){
      status ="Paid";
       flag =false;
      return"Payment Successful." ;

    else{
      flag =false;
      return "Ticket price is already paid!";
    }
  }
  
  public String toString(){
    return "Ticket ID: "+ticketId+"\n"+super.toString()+"\nGenre: "+genre+"\nSeat Type: " +seat + "\nPrice(tk): "+getPrice()+"\nStatus: "+status;
    
  }

public class KKTea extends Tea{
  public int bags;
  public int weight;
  public static int sales=0;
  public static int regularTea=0;
   public static int flavouredTea=0;
  public KKTea(int price,int bags){
    super("KK Regular Tea",price);
    this.weight =bags*2;
    this.bags =bags;
  }
  public void productDetail(){
    super.productDetail();
    System.out.println("Weight: "+weight+", Tea Bags: "+bags);
  }
  public static void totalSales(){
    if(flavouredTea==0){
    System.out.println("Total Sales: "+sales+"\nKK Regular Tea: "+regularTea);
    }
    else{
      System.out.println("Total Sales: "+sales+"\nKK Regular Tea: "+regularTea+"\nKK Flavoured Tea: "+flavouredTea);
    }
  }
  public static void updateSoldStatusRegular(KKTea k){
    k.status=true;
    sales++;
    regularTea++;
  }
}

  }
