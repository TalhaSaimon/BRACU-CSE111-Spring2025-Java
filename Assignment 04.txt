public class Triangle{
  public int s1,s2,s3,perimeter;
  public String type;
  public Triangle(int a ,int b, int c){
    s1=a;
    s2=b;
    s3=c;
   perimeter =s1+s2+s3;
  }
  public void triangleDetails(){
  
    System.out.println("Three sides of the triangle are: "+s1+", "+s2+", "+s3+"\nPerimeter: "+perimeter);
  }
  public String printTriangleType(){
    if(s1==s2 && s2==s3){
      type ="Equilateral";
     return "This is an "+type+" Triangle.";
    }
    else if(s1 != s2 && s2 != s3 && s1!=s3){
       type ="Scalene";
       return "This is a "+type+" Triangle.";
    }
    else{
       type ="Isosceles";
      return "This is an "+type+" Triangle.";
    }
  }
  public void compareTrinagles(Triangle c){
    if (c != this){
      if(s1==c.s1 && s2==c.s2 &&s3==c.s3){
        System.out.println("Addresses are different but the sides of the triangles are equal.");
      }
      else if(perimeter == c.perimeter){
        System.out.println("Only the perimeter of both triangles is equal.");
      }
      else{
        System.out.println("Addresses, length of the sides and perimeter all are different.");
      }
    }
    else{
    System.out.println("These two triangle objects have the same address.");
    }
  }
}

    passenger =new String [limit];
  }
  public BracuBus(String s, int n){
    route=s;
    limit =n;
    passenger =new String [limit];
  }
  public void showDetails(){
    System.out.println("Bus Route: "+route+"\nPassenger Count: "+idx+" (Max: "+limit+")\nPassengers on Board:");
    if(idx>0){
      for(int i=0;i<idx;i++){
        System.out.print(passenger[i] +" ");
      }
      System.out.println();
    }
  }
   public void board(){
    System.out.println("No passengers");
  }
   public void board(BracuStudent c){
    if(c.pass){
      if(idx<limit){
        if(c.location.equals(route)){
       passenger[idx++] =c.name;
       System.out.println(c.name+" boarded the bus.");
        }
        else{
          System.out.println("You got on the wrong bus!");
        }
      }
      else{
       System.out.println("Bus is full!");
     }
    }
    else{
    System.out.println("You don't have a bus pass!");
     }                      
  }
   public void board(BracuStudent c1,BracuStudent c2){
        board(c1);
        board(c2);
  }
