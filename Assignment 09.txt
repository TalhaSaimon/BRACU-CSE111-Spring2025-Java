public class Pokemon{
  public String attack;
  public int power;
  public String name;
  public Pokemon(String name){
    this.name =name;
  }
  public void attack(){
    System.out.println(name+" attacks with a basic move!");
  }
  public void attack(String attack){
    this.attack =attack;
    System.out.println(name+" uses "+attack+"!");
  }
  public void attack(String attack,int n){
    this.attack =attack;
    power =n;
    System.out.println(name+" uses "+attack+" with power "+power+"!");
  }
}

public class Car extends Vehicle{
  public String model;
  public int doors;
  public boolean ai;
  public Car(String brand, String model,int year,int doors,int wheel, boolean ai){
    super(brand,year);
    this.model =model;
     setWheels(wheel);
    this.doors=doors;
    this.ai=ai;
  }
  public String toString(){
    return super.toString()+", Model: "+model+", Doors: "+doors+", AI: "+ai;
  }
  public void startAutoPilot(){
    if(ai){
      System.out.println(getBrand() +":"+model+" AutoPilot Started");
    }
    else{
      System.out.println(getBrand() +":"+model+" has NO AutoPilot");
    }
  }
}

public class Garage{
  Car cars [];
  Bike bikes[];
  int idxCar=0;
  int idxBike=0;
  public Garage(int car,int bike){
    this.cars= new Car[car];
    this.bikes= new Bike[bike];
    System.out.println("Welcome to the Garage!\nCar Capacity: "+car+"\nBike Capacity: "+bike);
  }
  public void addVehicle(Vehicle v){
    if(v instanceof Car){
       
      if(idxCar<cars.length){
        cars[idxCar++] =(Car)v;
          System.out.println("A "+v.getBrand()+" CAR has been added to the Garage");
      }
      else{
       System.out.println("Can't add more Cars! Capacity: "+cars.length);
      }
    }
    else{
      v=(Bike)v;
      if(idxBike<bikes.length){
        bikes[idxBike++] =(Bike)v;
        System.out.println("A "+v.getBrand()+" BIKE has been added to the Garage");
      }
     
      else{
        System.out.println("Can't add more bikes! Capacity: "+bikes.length);
      }
        
    }
  }
  
}
