import java.util.Arrays;
import java.util.Scanner;
public class CellPhone{
  public String model = "unknown";
  public String arr[] = new String[3];
  public int idx=0;

  public void printDetails(){
    System.out.println("Phone Model "+model);
    System.out.println("Contacts Stored "+idx);
    if(idx!=0){
      System.out.println("Stored Contacts:");
      for(int i =0;i<idx;i++){
      System.out.println(arr[i]);
      
    }
    }
  }
  public void storeContact(String s1){
    if(idx <3){
      arr[idx++] =s1;
      System.out.println("Contact Stored");
    }
    else{
      System.out.println("Memory full. New contact can't be stored.");
    }
  }
}

public class Employee{
  public String name ="";
  public double salary = 30000.00;
  public String designation ="junior";
  public void newEmployee(String s1){
    name =s1;
  }
  public void displayInfo(){
   System.out.println("Employee Name: "+name);
   System.out.println("Employee Salary: "+salary+" Tk");
   System.out.println("Employee Designation: "+designation);
  }
  public void calculateTax(){
    if(salary<=30000){
      System.out.println("No need to pay tax");
    }
    else if(salary <50000){
      System.out.println(name+" Tax Amount: "+((salary*10)/100)+" Tk");
    }
    else{
      System.out.println(name+" Tax Amount: "+((salary*30)/100)+" Tk");
    }
  }
  public void promoteEmployee(String s){
    if(s.equals("senior")){
      salary = salary+25000.00;
      designation = "senior";
      System.out.println(name+" has been promoted to "+ designation);
      System.out.println("New Salary: "+ salary+" Tk");
    }
    else if(s.equals("lead")){
      salary = salary+50000.00;
      designation = "lead";
      System.out.println(name+" has been promoted to "+ designation);
      System.out.println("New Salary: "+ salary+" Tk");
    }
    else if(s.equals("manager")){
      salary = salary+75000.00;
      designation = "manager";
      System.out.println(name+" has been promoted to "+ designation);
      System.out.println("New Salary: "+ salary+" Tk");
    }
}
}
