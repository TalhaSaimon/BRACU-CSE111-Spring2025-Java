public class Cart{
  public String []items;
  public double [] price;
  public double discount=0.0;
  public double total_price=0.0;
  public int cart=0;
  int idx =0;
  public void create_cart(int n){
    cart=n;
    items = new String [3];
    price = new double [3];
  }
  public void addItem(String item, double price){
    if(idx<3){
    this.total_price += price;
    this.items[idx] = item;
    this.price[idx++] = price;
    System.out.println(item+" added to cart "+cart+"\nYou have "+idx+" item(s) in your cart now.");
    
    }
    else{
      System.out.println("You already have 3 items on your cart");
    }
  }
  public void addItem(double price, String item){
    addItem(item,price);
  }
  public void cartDetails(){
    System.out.println("Your cart(c"+cart+") :");
    for(int i =0;i<idx;i++){
      System.out.println(items[i]+" - "+price[i]);
      
    }
     System.out.println("Discount Applied: "+discount+"%\nTotal price: "+total_price);
  }
  public void giveDiscount(double n){
    this.discount =n;
    total_price -= total_price*(n/100);
  }
}

public class Reader{
  public String name ="New user";
  public int capacity = 0;
  public String[]books;
  int idx =0;
  public String createReader(String name,int n){
    this.name =name;
    this.capacity =n;
    books =new String[capacity];
    return "A new reader is created!";
  }
  public void addBook(String book){
   if(idx<capacity){
     this.books[idx++]=book;
    }
   else{
   System.out.println("No more capacity");
   }
  }
  public void readerInfo(){
    System.out.println("Name: "+name+"\nCapacity: "+capacity +"\nBooks:");
    if(idx ==0){
      System.out.println("No books added yet");
    }
    else{
      for(int i =0; i<idx;i++){
        System.out.println("Book "+(i+1)+": "+books[i]);
      }
    }
  }
}
